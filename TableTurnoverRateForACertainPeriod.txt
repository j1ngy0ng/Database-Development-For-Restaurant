SET linesize  120
SET pagesize 35

ACCEPT start_date CHAR PROMPT 'Enter the start date (YYYY-MM-DD): '
ACCEPT end_date CHAR PROMPT 'Enter the end date (YYYY-MM-DD): '

COLUMN TableNo FORMAT 999 HEADING "Table Number"
COLUMN TotalOrders FORMAT 999,999 HEADING "Total Orders"
COLUMN TotalaCustomers FORMAT 999,999 HEADING "Total Customers"
COLUMN TurnoverRate FORMAT 999.9999 HEADING "Table Turnover Rate"

TTITLE CENTER 'Table Turnover Rate for A Certain Period' SKIP 3-
LEFT 'Start Date: &start_date' SKIP 1-
RIGHT 'End Date: &end_date' SKIP 2-

BREAK ON REPORT SKIP 1
COMPUTE SUM OF TotalOrders ON REPORT
COMPUTE SUM OF TotalCustomers ON REPORT
COMPUTE SUM OF TurnoverRate ON REPORT

SELECT rt.TableNo, COUNT(fo.OrderNo) AS TotalOrders, SUM(fo.NoOfCust) AS TotalCustomers, COUNT(fo.OrderNo) / (SELECT COUNT(DISTINCT TableNo) FROM RESTAURANT_TABLE) AS TurnoverRate
FROM RESTAURANT_TABLE rt
JOIN FOOD_ORDER fo ON rt.TableNo = fo.TableNo
WHERE fo.OrderDate BETWEEN TO_DATE('&start_date', 'YYYY-MM-DD') AND TO_DATE('&end_date', 'YYYY-MM-DD')
GROUP BY rt.TableNo
ORDER BY rt.TableNo;

CLEAR COLUMNS
CLEAR BREAKS
CLEAR COMPUTES
TTITLE OFF
