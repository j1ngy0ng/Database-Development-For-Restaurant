SET linesize 120
SET pagesize 35

ACCEPT start_date CHAR PROMPT 'Enter the start date (YYYY-MM-DD): '
ACCEPT end_date CHAR PROMPT 'Enter the end date (YYYY-MM-DD): '

COLUMN TableNo FORMAT 99 HEADING "Table No"
COLUMN Capacity FORMAT 99 HEADING "Capacity"
COLUMN Location FORMAT A20 HEADING "Location"

TTITLE CENTER  'Total Orders by Table Number' SKIP 2-
       LEFT 'Start Date: &start_date' SKIP 1-
       LEFT 'End Date: &end_date' SKIP 2-

SELECT rt.TableNo, rt.Capacity, rt.Location, COUNT(*) AS "Total Orders"
FROM RESTAURANT_TABLE rt 
JOIN FOOD_ORDER fo ON rt.TableNo = fo.TableNo
WHERE fo.OrderDate BETWEEN TO_DATE('&start_date', 'YYYY-MM-DD') AND TO_DATE('&end_date', 'YYYY-MM-DD')
GROUP BY rt.TableNo, rt.Capacity, rt.Location
ORDER BY rt.TableNo;

CLEAR COLUMNS
CLEAR BREAKS
CLEAR COMPUTES
TTITLE OFF
